// 35. Write a C# program to find and display the longest word in a given string. 

using System;

class third
{
    static void Main(string[] args)
    {
        string text = "Find the longest sssssword";
        string longest = "";
        
        string[] words = text.Split(' ');
        
        for (int i = 0; i < words.Length; i++)
        {
            if (words[i].Length > longest.Length)
            {
                longest = words[i];
            }
        }

        Console.WriteLine(longest);
    }
}
