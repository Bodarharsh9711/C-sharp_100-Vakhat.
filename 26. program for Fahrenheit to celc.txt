// 26. program for Fahrenheit to celcious 

using System;

namespace first
{
    class first
    {	
        static void Main(string[] args)
        {
            Console.Write("Enter your number (Celsius): ");    
            double cal = Convert.ToDouble(Console.ReadLine());
            double far = (cal * (9.0 / 5.0)) + 32;
            Console.WriteLine($"Temperature in Fahrenheit: {far}");      
        }
    }
}
